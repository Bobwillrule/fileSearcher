cmake_minimum_required(VERSION 3.25)
project(FileSearcher VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FILESEARCHER_BUILD_UI "Build the Qt desktop shell" ON)
option(FILESEARCHER_MODEL_TESTS "Enable pinned BGE parity tests (requires prepared model)" OFF)
set(ONNXRUNTIME_ROOT "${CMAKE_SOURCE_DIR}/.tools/onnxruntime/onnxruntime-win-x64-1.23.2" CACHE PATH "Official ORT 1.23.2 x64 package")
include(CTest)

find_package(hnswlib CONFIG REQUIRED)
add_library(filesearcher_core src/search/vector_index.cpp)
target_include_directories(filesearcher_core PUBLIC src)
target_link_libraries(filesearcher_core PRIVATE hnswlib::hnswlib)
if(MSVC)
  target_compile_options(filesearcher_core PRIVATE /W4 /permissive- /utf-8)
endif()

if(BUILD_TESTING)
  find_package(Catch2 3 CONFIG REQUIRED)
  add_executable(filesearcher_tests tests/vector_index_tests.cpp)
  target_link_libraries(filesearcher_tests PRIVATE filesearcher_core Catch2::Catch2WithMain)
  include(Catch)
  catch_discover_tests(filesearcher_tests)
endif()

if(FILESEARCHER_BUILD_UI)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Quick QuickControls2 Pdf Sql Concurrent)
  find_package(libzip CONFIG REQUIRED)
  qt_standard_project_setup(REQUIRES 6.8)
  add_library(content_text src/extraction/passages.cpp)
  target_include_directories(content_text PUBLIC src)
  target_link_libraries(content_text PUBLIC Qt6::Core)
  add_library(metadata_core src/storage/metadata_store.cpp src/indexing/file_scanner.cpp src/indexing/work_queue.cpp src/indexing/directory_watcher.cpp src/indexing/directory_watcher.h)
  target_include_directories(metadata_core PUBLIC src)
  target_link_libraries(metadata_core PUBLIC Qt6::Core Qt6::Sql content_text)
  add_subdirectory(src/app)
  if(BUILD_TESTING)
    add_executable(metadata_tests tests/metadata_tests.cpp)
    target_link_libraries(metadata_tests PRIVATE metadata_core Catch2::Catch2 advapi32)
    add_test(NAME metadata_indexing COMMAND metadata_tests)
    set_tests_properties(metadata_indexing PROPERTIES TIMEOUT 90)
    add_executable(app_controller_tests tests/app_controller_tests.cpp tests/incremental_tests.cpp tests/app_content_tests.cpp src/app/app_controller.cpp src/app/app_controller.h)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
    add_executable(qml_workflow_tests tests/qml_workflow_tests.cpp src/app/app_controller.cpp src/app/app_controller.h)
    target_link_libraries(qml_workflow_tests PRIVATE metadata_core content_indexer embedding_indexer Qt6::Quick Qt6::QuickControls2 Qt6::Test Qt6::Concurrent shell32 ole32)
    add_test(NAME keyboard_storage_ui COMMAND qml_workflow_tests "${CMAKE_SOURCE_DIR}/src/app/Main.qml" "${CMAKE_BINARY_DIR}/ui-checks")
    set_tests_properties(keyboard_storage_ui PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QUICK_BACKEND=software;QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts")
    target_link_libraries(app_controller_tests PRIVATE metadata_core content_indexer embedding_indexer Catch2::Catch2 Qt6::Concurrent shell32 ole32)
    add_test(NAME desktop_search_workflow COMMAND app_controller_tests)
    set_tests_properties(desktop_search_workflow PROPERTIES TIMEOUT 60)
    add_executable(metadata_probe tools/metadata_probe.cpp)
    target_link_libraries(metadata_probe PRIVATE metadata_core)
  endif()
  add_library(extraction_runner src/extraction/probe_runner.cpp)
  target_include_directories(extraction_runner PUBLIC src)
  target_link_libraries(extraction_runner PUBLIC Qt6::Core)
  add_library(content_indexer src/indexing/content_indexer.cpp)
  target_link_libraries(content_indexer PUBLIC metadata_core extraction_runner)
  add_executable(extraction_probe tools/extraction_probe.cpp)
  target_link_libraries(extraction_probe PRIVATE extraction_runner)
  if(WIN32)
    add_executable(ifilter_worker src/extraction/ifilter_worker.cpp)
    target_include_directories(ifilter_worker PRIVATE src)
    target_link_libraries(ifilter_worker PRIVATE Qt6::Core ole32 ntquery)
    add_executable(document_worker src/extraction/document_worker.cpp)
    target_include_directories(document_worker PRIVATE src)
    target_link_libraries(document_worker PRIVATE Qt6::Pdf Qt6::Gui libzip::zip content_text)
    add_dependencies(filesearcher document_worker)
    if(BUILD_TESTING)
      add_dependencies(app_controller_tests document_worker)
    endif()
  endif()
  if(BUILD_TESTING)
    add_executable(probe_fixture_worker tests/probe_fixture_worker.cpp)
    target_link_libraries(probe_fixture_worker PRIVATE Qt6::Core)
    target_include_directories(probe_fixture_worker PRIVATE src)
    add_executable(probe_runner_tests tests/probe_runner_tests.cpp)
    target_link_libraries(probe_runner_tests PRIVATE extraction_runner)
    add_test(NAME extraction_worker_isolation COMMAND probe_runner_tests $<TARGET_FILE:probe_fixture_worker>)
    set_tests_properties(extraction_worker_isolation PROPERTIES TIMEOUT 15)
    if(WIN32)
      if(EXISTS "${CMAKE_SOURCE_DIR}/.tools/python/Scripts/python.exe")
        set(Python3_EXECUTABLE "${CMAKE_SOURCE_DIR}/.tools/python/Scripts/python.exe")
      endif()
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME prepare_document_fixtures COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/create_extraction_fixtures.py")
      set_tests_properties(prepare_document_fixtures PROPERTIES FIXTURES_SETUP document_fixtures)
      add_executable(document_tests tests/document_tests.cpp)
      target_link_libraries(document_tests PRIVATE extraction_runner)
      add_test(NAME bundled_document_coverage COMMAND document_tests $<TARGET_FILE:document_worker> "${CMAKE_SOURCE_DIR}/build/fixtures")
      set_tests_properties(bundled_document_coverage PROPERTIES FIXTURES_REQUIRED document_fixtures TIMEOUT 60)
      add_executable(content_tests tests/content_tests.cpp)
      target_link_libraries(content_tests PRIVATE content_indexer Catch2::Catch2)
      add_dependencies(content_tests document_worker probe_fixture_worker)
      add_test(NAME content_search COMMAND content_tests "${CMAKE_SOURCE_DIR}/build/fixtures")
      set_tests_properties(content_search PROPERTIES FIXTURES_REQUIRED document_fixtures TIMEOUT 90)
      add_executable(ifilter_smoke_tests tests/ifilter_smoke_tests.cpp)
      target_link_libraries(ifilter_smoke_tests PRIVATE extraction_runner)
      add_test(NAME windows_text_extraction COMMAND ifilter_smoke_tests $<TARGET_FILE:ifilter_worker> "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/notes.txt")
      set_tests_properties(windows_text_extraction PROPERTIES TIMEOUT 15)
    endif()
  endif()
  add_library(embedding_support src/ai/bert_tokenizer.cpp src/ai/model_assets.cpp src/ai/embedding_client.cpp)
  target_include_directories(embedding_support PUBLIC src)
  target_link_libraries(embedding_support PUBLIC Qt6::Core)
  add_library(embedding_indexer src/storage/embedding_store.cpp src/search/vector_cache.cpp src/indexing/embedding_indexer.cpp src/search/search_store.cpp src/search/hybrid_search.cpp)
  target_link_libraries(embedding_indexer PUBLIC metadata_core embedding_support filesearcher_core)
  if(EXISTS "${ONNXRUNTIME_ROOT}/lib/onnxruntime.lib" AND EXISTS "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll")
    add_library(onnxruntime SHARED IMPORTED)
    set_target_properties(onnxruntime PROPERTIES IMPORTED_IMPLIB "${ONNXRUNTIME_ROOT}/lib/onnxruntime.lib" IMPORTED_LOCATION "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" INTERFACE_INCLUDE_DIRECTORIES "${ONNXRUNTIME_ROOT}/include")
    add_library(embedding_engine src/ai/embedding_engine.cpp)
    target_link_libraries(embedding_engine PUBLIC embedding_support PRIVATE onnxruntime)
    add_executable(embedding_worker src/ai/embedding_worker.cpp)
    target_link_libraries(embedding_worker PRIVATE embedding_engine)
    add_custom_command(TARGET embedding_worker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:embedding_worker>")
    add_dependencies(filesearcher embedding_worker)
    if(EXISTS "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/onnx/model.onnx" AND EXISTS "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/vocab.txt" AND EXISTS "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/README.md")
      add_custom_target(embedding_assets ALL
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/models/bge-small-en-v1.5/onnx"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/onnx/model.onnx" "${CMAKE_BINARY_DIR}/models/bge-small-en-v1.5/onnx/model.onnx"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/vocab.txt" "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5/README.md" "${CMAKE_BINARY_DIR}/models/bge-small-en-v1.5"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/docs/licenses/BGE-MIT.txt" "${CMAKE_BINARY_DIR}/models/bge-small-en-v1.5/LICENSE"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/tools/model-assets.lock.json" "${CMAKE_BINARY_DIR}/models/bge-small-en-v1.5/model-assets.lock.json")
      add_dependencies(filesearcher embedding_assets)
    endif()
    if(FILESEARCHER_MODEL_TESTS)
      if(NOT TARGET embedding_assets)
        message(FATAL_ERROR "Prepare the pinned model assets before enabling model tests")
      endif()
      add_executable(embedding_probe tools/embedding_probe.cpp)
      target_link_libraries(embedding_probe PRIVATE embedding_engine filesearcher_core)
      add_dependencies(embedding_probe embedding_worker)
      add_test(NAME bge_reference_parity COMMAND embedding_probe "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5" "${CMAKE_SOURCE_DIR}/tools/model-assets.lock.json")
      set_tests_properties(bge_reference_parity PROPERTIES TIMEOUT 180)
    endif()
  elseif(FILESEARCHER_MODEL_TESTS)
    message(FATAL_ERROR "Prepare ONNX Runtime before enabling model tests")
  else()
    message(STATUS "ONNX Runtime unavailable: building lexical search with AI-unavailable fallback")
  endif()
  if(BUILD_TESTING)
    add_executable(embedding_tests tests/embedding_tests.cpp)
    add_executable(hybrid_tests tests/hybrid_tests.cpp)
    target_link_libraries(hybrid_tests PRIVATE embedding_indexer Catch2::Catch2)
    add_test(NAME hybrid_ranking_filters COMMAND hybrid_tests)
    set_tests_properties(hybrid_ranking_filters PROPERTIES TIMEOUT 90)
    target_link_libraries(embedding_tests PRIVATE embedding_indexer Catch2::Catch2)
    add_dependencies(embedding_tests probe_fixture_worker)
    add_test(NAME embedding_storage_recovery COMMAND embedding_tests)
    set_tests_properties(embedding_storage_recovery PROPERTIES TIMEOUT 60)
    if(FILESEARCHER_MODEL_TESTS)
      add_executable(retrieval_eval tools/retrieval_eval.cpp)
      add_executable(search_benchmark tools/search_benchmark.cpp)
      add_executable(package_smoke tools/package_smoke.cpp src/app/app_controller.cpp src/app/app_controller.h)
      target_link_libraries(package_smoke PRIVATE metadata_core content_indexer embedding_indexer Qt6::Concurrent shell32 ole32)
      add_dependencies(package_smoke document_worker embedding_worker embedding_assets)
      target_link_libraries(search_benchmark PRIVATE embedding_indexer psapi)
      add_dependencies(search_benchmark embedding_worker embedding_assets)
      target_link_libraries(retrieval_eval PRIVATE embedding_indexer)
      add_dependencies(retrieval_eval embedding_worker embedding_assets)
      add_test(NAME hybrid_retrieval_quality COMMAND retrieval_eval "${CMAKE_SOURCE_DIR}/tests/fixtures/hybrid_retrieval.json" "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5" all "${CMAKE_BINARY_DIR}/retrieval-evaluation.json")
      set_tests_properties(hybrid_retrieval_quality PROPERTIES TIMEOUT 300)
      add_dependencies(embedding_tests embedding_worker embedding_assets)
      add_dependencies(app_controller_tests embedding_worker embedding_assets)
      add_test(NAME desktop_embedding_workflow COMMAND app_controller_tests "[.model]")
      set_tests_properties(desktop_embedding_workflow PROPERTIES TIMEOUT 60)
      add_test(NAME local_embedding_workflow COMMAND embedding_tests --model "${CMAKE_SOURCE_DIR}/models/bge-small-en-v1.5")
      set_tests_properties(local_embedding_workflow PROPERTIES TIMEOUT 180)
    endif()
  endif()
endif()
